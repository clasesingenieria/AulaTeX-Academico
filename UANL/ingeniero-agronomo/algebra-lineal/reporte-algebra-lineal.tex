\begin{abstractd}
Este portafolio formal re\'{u}ne las evidencias y productos acad\'{e}micos de la unidad de aprendizaje \'{A}lgebra Lineal del programa educativo Ingeniero Agr\'{o}nomo de la Facultad de Agronom\'{i}a de la Universidad Aut\'{o}noma de Nuevo Le\'{o}n. En \'{e}l se integran las competencias desarrolladas a lo largo de las tres fases del curso y el Producto Integrador de Aprendizaje (PIA): representaci\'{o}n matricial y determinantes, resoluci\'{o}n y an\'{a}lisis de sistemas de ecuaciones lineales aplicados a balances de nutrientes y fen\'{o}menos biol\'{o}gicos, teor\'{i}a espectral con autovalores y autovectores en matrices de Leslie, y modelaci\'{o}n matem\'{a}tica cuantitativa orientada al sector agron\'{o}mico.
\end{abstractd}

\section{Estructura Curricular y Portafolio de Evidencias}

\subsection{Fase 1: Matrices y Determinantes}
\begin{itemize}
  \item \textbf{Actividad 1 (Evidencia 1 / Actividad 1.6):} Mapa cognitivo de telara\~{n}a sobre aplicaciones del \'{a}lgebra lineal en la vida cotidiana y agronom\'{i}a. Ponderaci\'{o}n: 10 pts.
\end{itemize}

\subsection{Fase 2: Sistemas de Ecuaciones Lineales}
\begin{itemize}
  \item \textbf{Actividad 2 (Laboratorio 2):} Segundo Laboratorio de \'Algebra Lineal. Resoluci\'on completa de sistemas lineales de $3\times 3$ y $4\times 4$ mediante m\'etodos de eliminaci\'on de Gauss, Gauss-Jordan y regla de Cramer. Ponderaci\'on: 3 pts.
  \item \textbf{Actividad 2.6 (Evidencia 2):} Mapa cognitivo de algoritmo y resoluci\'{o}n paso a paso del Sistema 11 con verificaci\'{o}n de independencia lineal y comprobaci\'{o}n por Cramer. Ponderaci\'{o}n: 10 pts.
\end{itemize}

\subsection{Fase 3: Valores y Vectores Propios}
\begin{itemize}
  \item \textbf{Actividad 3.6 (Evidencia 3):} Reporte t\'{e}cnico de la soluci\'{o}n de un caso de aplicaci\'{o}n de los conceptos de valores y vectores propios en din\'{a}micas poblacionales y dispersi\'{o}n agron\'{o}mica (Sistema 11). Ponderaci\'{o}n: 10 pts.
\end{itemize}

\subsection{Producto Integrador de Aprendizaje (PIA)}
\begin{itemize}
  \item \textbf{Reporte T\'{e}cnico Institucional del PIA:} Modelaci\'{o}n matricial de fertilizaci\'{o}n balanceada N-P-K y din\'{a}mica poblacional de vectores citr\'{i}colas (\textit{Diaphorina citri}) mediante matrices de Leslie y an\'{a}lisis espectral. Ponderaci\'{o}n: 25 pts.
  \item \textbf{Presentaci\'{o}n Ejecutiva Beamer (16:9):} Sustentaci\'{o}n oral institucional para defensa del PIA ante la comisi\'{o}n docente.
\end{itemize}
