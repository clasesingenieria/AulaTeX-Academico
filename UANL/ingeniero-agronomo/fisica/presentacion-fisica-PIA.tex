\begin{frame}{1. Justificación del Proyecto en la Agronomía}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}¿Por qué una excavadora para un agrónomo?}
      \begin{itemize}
        \item \textbf{Zanjas de drenaje parcelario:} Para evitar encharcamientos y asfixia de raíces en suelos arcillosos de Nuevo León.
        \item \textbf{Instalación de riego por goteo:} Para enterrar las mangueras maestras protegiéndolas del sol y maquinaria.
        \item \textbf{Movimiento de composta y abonos:} Cargar y voltear sustratos pesados sin esfuerzo manual excesivo.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}El Principio Físico en el Campo:}
      \begin{itemize}
        \item Toda retroexcavadora o pala frontal de tractor opera con cilindros y fluido a presión.
        \item El prototipo demuestra que con leyes físicas sencillas podemos multiplicar la fuerza aplicada con los dedos.
        \item Se diseñó como un prototipo funcional construido con materiales caseros y reciclados.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2. Fundamento Físico: El Principio de Pascal}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Definición de Presión:}
      $$P = \frac{F}{A} \quad \left[\frac{\text{N}}{\text{m}^2} = \text{Pa}\right]$$
      \begin{itemize}
        \item La fuerza aplicada se distribuye sobre el área del émbolo.
        \item Los líquidos son \textbf{incompresibles}: no se reducen de volumen al presionarlos.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Principio de Pascal (Prensa Hidráulica):}
      $$P_1 = P_2 \implies \frac{F_1}{A_1} = \frac{F_2}{A_2} \implies F_2 = F_1 \left(\frac{A_2}{A_1}\right)$$
      \begin{itemize}
        \item Al empujar una jeringa chica ($10\text{ mL}$), la jeringa grande ($20\text{ mL}$) genera una fuerza mayor para levantar tierra.
        \item \textbf{Purga obligatoria:} Si queda aire, el aire se comprime y la máquina pierde fuerza y firmeza.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{3. Materiales Caseros y Proceso de Construcción}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Materiales Empleados (Disponibles en Casa):}
      \begin{itemize}
        \item Listones de madera reciclada cortados al plano:
          \begin{itemize}
            \item Base y chasis: $10.5\text{ cm}$.
            \item Mecanismo de cuchara: $17.5\text{ cm}$.
            \item Pluma principal: $26.0\text{ cm}$.
            \item Antebrazo de alcance: $36.0\text{ cm}$.
          \end{itemize}
        \item 8 jeringas plásticas ($4$ de $10\text{ mL}$ y $4$ de $20\text{ mL}$).
        \item $3\text{ m}$ de manguera flexible delgada transparente.
        \item Agua de la llave teñida con colorante vegetal.
        \item Cinchos de plástico y tornillos pequeños con tuerca.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}4 Grados de Libertad (Movimientos):}
      \begin{enumerate}
        \item \textbf{Giro de base ($\approx 170^\circ$):} Mueve la máquina hacia los lados para descargar.
        \item \textbf{Elevación de pluma ($60^\circ$):} Sube y baja todo el brazo de madera.
        \item \textbf{Extensión de antebrazo ($80^\circ$):} Acerca o aleja la pala del suelo.
        \item \textbf{Cuchara excavadora ($90^\circ$):} Abre y cierra la pala para atrapar la tierra.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{4. Resultados Esperados vs. Obtenidos}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Resultados Esperados:}
      \begin{itemize}
        \item Que las 4 articulaciones se movieran sin atorarse.
        \item Multiplicar la fuerza de los dedos sin romper la madera.
        \item Que no hubiera fugas de agua.
        \item Levantar muestras de tierra (\SI{50}{a} \SI{100}{g}).
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Resultados Obtenidos en Pruebas:}
      \begin{itemize}
        \item Movimiento suave e instantáneo al purgar todo el aire.
        \item La pala levantó hasta \textbf{\SI{120}{g}} de tierra húmeda de maceta sin esfuerzo.
        \item Se realizaron más de 30 ciclos completos de zanjado sin ninguna fuga ni falla de mangueras.
        \item Se comprobó empíricamente que el agua incompresible transmite la presión de manera inmediata.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
