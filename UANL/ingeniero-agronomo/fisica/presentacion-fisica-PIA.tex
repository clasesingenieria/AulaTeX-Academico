\begin{frame}{1. Justificación y Problemática Agronómica}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}El Reto del Manejo de Suelos Agrícolas:}
      \begin{itemize}
        \item Los suelos del norte de México demandan zanjado parcelario para evitar la anoxia radicular y salinización freática.
        \item La apertura de zanjas para tendido de líneas presurizadas de fertirriego requiere maquinaria de alta potencia y precisión.
        \item Toda maquinaria pesada de campo (retroexcavadoras, palas frontales) opera bajo circuitos de fluidos presurizados.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}Solución Física: Principio de Pascal:}
      \begin{itemize}
        \item Transmisión isotrópica e instantánea de la presión hidrostática en fluidos incompresibles:
          $$P = \frac{F_1}{A_1} = \frac{F_2}{A_2}$$
        \item Multiplicación cuadrática de fuerza mediante émbolos diferenciales:
          $$F_2 = F_1 \left(\frac{D_2}{D_1}\right)^2$$
        \item Construcción física tangible basada en el plano industrial \textit{Science Technology Project Teaching -- Excavating machinery}.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{2. Formulación Analítica y Leyes Fundamentales}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}1. Hidrostática y Continuidad Volumétrica:}
      \begin{itemize}
        \item Fluido incompresible ($\rho \approx \text{constante}$):
          $$\Delta V_1 = \Delta V_2 \implies A_1 \Delta x_1 = A_2 \Delta x_2$$
        \item Desplazamiento del émbolo de salida:
          $$\Delta x_2 = \Delta x_1 \left(\frac{D_1}{D_2}\right)^2$$
        \item Conservación del trabajo mecánico ideal:
          $$W = F_1 \Delta x_1 = F_2 \Delta x_2$$
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{uanlblue}2. Rendimiento Real y Momentos de Fuerza:}
      \begin{itemize}
        \item Ventaja mecánica ideal vs real:
          $$VM_{\text{ideal}} = \frac{A_2}{A_1} = 2.25, \quad VM_{\text{real}} = \frac{F_{2,\text{med}}}{F_1}$$
        \item Eficiencia mecánica con fricción de sellos:
          $$\eta_{\text{mec}} = \frac{VM_{\text{real}}}{VM_{\text{ideal}}} = 90.7\%$$
        \item Equilibrio rotacional estático en pasadores:
          $$\sum \tau_O = F_{\text{hid}} \cdot d_{\perp} - W_{\text{brazo}} r_{\text{cm}} - F_{\text{carga}} L = 0$$
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

      \begin{enumerate}
        \item \textbf{GDL 1 (Rotación de Base):} Giro azimutal sobre corona ($\Delta \theta = 180^\circ$).
        \item \textbf{GDL 2 (Elevación de Pluma):} Inclinación vertical del aguilón ($\Delta \alpha = 65^\circ$).
        \item \textbf{GDL 3 (Articulación Antebrazo):} Alcance y repliegue ($\Delta \beta = 85^\circ$).
        \item \textbf{GDL 4 (Vuelco de Cuchara):} Penetración, desgarre y vaciado ($\Delta \gamma = 95^\circ$).
      \end{enumerate}
    \end{column}
  \end{columns}
