\begin{frame}{Modelar antes de programar}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La sesión integra sistemas, perímetros y funciones.
\item Una solución teórica debe contrastarse con la ejecución física.
\item Fricción, batería y superficie pueden cambiar la calibración.
\end{itemize}
\end{frame}
\begin{frame}{Sistema propuesto}
\fontsize{14}{18}\selectfont\[\begin{cases}x+y=36\\2x+y=50\end{cases}\qquad\Rightarrow\qquad x=14,\quad y=22.\]
\[14+22=36,\qquad 2(14)+22=50.\]
\vspace{.3cm}Es un ejemplo propio, no una transcripción de un sistema ausente. Los parámetros de avance y giro requieren calibraciones distintas.
\end{frame}
\begin{frame}{Pista de trapecio}
\fontsize{14}{18}\selectfont\[P=(b+4)+b+5+5=2b+14.\]
\[P=30\Rightarrow b=8,\quad h=\sqrt{5^2-2^2}=\sqrt{21}.\]
\[\alpha=\arccos(2/5)\approx66.42^\circ.\]
\small Giros exteriores: $113.58^\circ,66.42^\circ,66.42^\circ,113.58^\circ$. Suman $360^\circ$; la prueba física contrasta el cierre ideal.
\end{frame}
\begin{frame}{Función de calibración}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Ejemplo simulado: 2, 4 y 6 s producen 30, 60 y 90 cm.
\item d = 15t; para 120 cm se requieren 8 s.
\item Si aparece un término inicial, revisar origen, retardo y sesgo.
\end{itemize}
\end{frame}
\begin{frame}{Secuencia y participación}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Definir variables y criterios; programar, medir y revisar.
\item Rotar programación, registro, medición y explicación.
\item Utilizar simulador o cuadrícula cuando falte equipo.
\end{itemize}
\end{frame}
\begin{frame}{Reflexión y evidencia}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La robótica debe hacer visible una relación matemática.
\item Conservar primera solución y versión corregida.
\item Las pruebas disponibles se analizan cualitativamente.
\end{itemize}
\end{frame}
