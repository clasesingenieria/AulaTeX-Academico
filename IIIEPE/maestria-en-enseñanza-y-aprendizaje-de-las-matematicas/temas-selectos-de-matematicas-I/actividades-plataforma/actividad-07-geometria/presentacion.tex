\begin{frame}{Figuras planas y cuerpos}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Abrir un envase permite relacionar caras y desarrollo plano.
\item Un rectángulo que gira alrededor de un lado genera un cilindro.
\item Un triángulo rectángulo que gira alrededor de un cateto genera un cono.
\end{itemize}
\end{frame}
\begin{frame}{Cilindro y desarrollo}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La superficie lateral es un rectángulo de base 2$\pi$r y altura h.
\item Con r = 3 y h = 10 cm: área lateral = 60$\pi$ cm$^2$.
\item Con dos bases: área total = 78$\pi$ cm$^2$.
\end{itemize}
\end{frame}
\begin{frame}{Prisma y pirámide}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Prisma de 8 $\times$ 5 $\times$ 3 cm: área total = 158 cm$^2$.
\item Pirámide cuadrada de lado 6 y altura 4 cm: apotema = 5 cm.
\item Área lateral de la pirámide = 60 cm$^2$; total = 96 cm$^2$.
\end{itemize}
\end{frame}
\begin{frame}{Cono}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Con r = 3 y h = 4 cm, la generatriz mide 5 cm.
\item El sector lateral tiene radio 5 cm y ángulo 216$^\circ$.
\item Área lateral = 15$\pi$; total = 24$\pi$ cm$^2$.
\end{itemize}
\end{frame}
\begin{frame}{Diseño y comprobación}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Relacionar cada cara con un término del cálculo.
\item Separar pestañas, tolerancias y pérdidas de corte del modelo ideal.
\item Los cálculos no acreditan que el prototipo haya sido fabricado.
\end{itemize}
\end{frame}
\begin{frame}{Reflexión docente}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Usar cartón, dibujo y recursos digitales opcionales.
\item Evaluar correspondencia geométrica, cálculo y explicación.
\item Distribuir herramientas y decisiones entre todos los integrantes.
\end{itemize}
\end{frame}
