\begin{frame}{Generalizar una relación}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Una variable representa una cantidad definida.
\item Una expresión puede evaluarse; una ecuación afirma igualdad.
\item El reto del recurso: x + 123 = 258 $\rightarrow$ x = 135.
\end{itemize}
\end{frame}
\begin{frame}{Traducir con precisión}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item El triple de un número más cinco: 3x + 5.
\item La tercera parte de una suma: (x + 5)/3.
\item El cuadrado de una diferencia no es una diferencia de cuadrados.
\end{itemize}
\end{frame}

\begin{frame}{Trapecio y condiciones}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Bases x + 6 y x + 2; lados iguales x: P = 4x + 8.
\item Para x = 5, el perímetro es 28 cm.
\item La geometría no degenerada requiere x > 2.
\end{itemize}
\end{frame}
