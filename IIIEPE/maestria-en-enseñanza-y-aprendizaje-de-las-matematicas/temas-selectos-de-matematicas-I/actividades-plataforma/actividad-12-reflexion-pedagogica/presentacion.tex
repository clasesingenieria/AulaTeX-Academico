\begin{frame}{Mi posición docente}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La formación industrial aporta objetivos, secuencia y verificación.
\item Mi acercamiento humanista incorpora participación, contexto y revisión.
\item Un producto funcional no demuestra por sí solo comprensión de todos.
\end{itemize}
\end{frame}
\begin{frame}{Coherencia del proyecto STEAM}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Plantear un reto, indagar, organizar, comunicar y reflexionar.
\item Cada fase debe producir evidencia ligada al contenido matemático.
\item La revisión de Martín-Cudero et al. (2024) pide fortalecer fundamentación e implementación.
\end{itemize}
\end{frame}
\begin{frame}{Dominio matemático}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Racionales: equivalencia y sentido de operaciones.
\item Funciones y ecuaciones: constantes, condiciones e igualdad.
\item Geometría, estadística y robótica: unidades, variación y modelos.
\end{itemize}
\end{frame}
\begin{frame}{Ejes articuladores}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La consigna menciona cinco; el marco curricular reconoce siete.
\item Destacar inclusión, pensamiento crítico, interculturalidad, igualdad y vida saludable.
\item Integrar también lectura y escritura, y artes y experiencias estéticas.
\end{itemize}
\end{frame}
\begin{frame}{Evaluación y mejora}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item La evidencia debe cambiar el apoyo y permitir una revisión.
\item Boström y Palm (2023) no encontraron diferencias significativas en su intervención.
\item No prometer mejora solo por añadir actividades llamadas formativas.
\end{itemize}
\end{frame}
\begin{frame}{Dos ajustes viables}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Ciclo de predicción, resolución, devolución y nueva resolución.
\item Distintas representaciones con un criterio matemático común.
\item Evaluar interacciones y cambios en el tiempo, sin inventar resultados de aplicación.
\end{itemize}
\end{frame}
