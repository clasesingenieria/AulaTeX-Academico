% SLIDE 02: FASE STEAM 1 - RETO DETONADOR
\begin{frame}{Fase 1 STEAM · Saberes Comunitarios y Reto de Mantenimiento}
\small
\textbf{Contexto Operativo Industrial:} En una celda de manufactura, dos cuadrillas técnicas realizan $n=6$ ajustes preventivos a un mecanismo crítico. Se registran los tiempos (minutos):
\vspace{0.25cm}
\begin{columns}[T]
	\begin{column}{0.48\textwidth}
		\begin{block}{Cuadrilla A (Proceso Alfa)}
			\centering \textbf{Datos:} 8, 9, 10, 10, 11, 12 min\\[0.1cm]
			$\bar{x}_A = \frac{60}{6} = 10.0$ min
		\end{block}
	\end{column}
	\begin{column}{0.48\textwidth}
		\begin{block}{Cuadrilla B (Proceso Beta)}
			\centering \textbf{Datos:} 5, 7, 10, 10, 13, 15 min\\[0.1cm]
			$\bar{x}_B = \frac{60}{6} = 10.0$ min
		\end{block}
	\end{column}
\end{columns}
\vspace{0.35cm}
\begin{alertblock}{Pregunta Detonadora para la Comunidad de Aprendizaje}
	Si ambos equipos promedian exactamente \textbf{10 minutos por ajuste}, ¿basta el promedio aritmético para decidir qué cuadrilla ofrece mayor confiabilidad y menor riesgo operativo?
\end{alertblock}
\end{frame}

% SLIDE 03: PARADOJA DE LA TENDENCIA CENTRAL
\begin{frame}{Paradoja de la Tendencia Central}
\small
Al calcular exclusivamente las medidas de centralidad, observamos una coincidencia absoluta:
\vspace{0.2cm}
\renewcommand{\arraystretch}{1.2}
\begin{tabularx}{\linewidth}{X|c|c|>{\raggedright\arraybackslash}X}
\toprule
\rowcolor{iiiepePale} \textbf{Medida Estadística} & \textbf{Cuadrilla A} & \textbf{Cuadrilla B} & \textbf{Interpretación Didáctica} \\
\midrule
\textbf{Media Aritmética ($\bar{x}$)} & 10.0 min & 10.0 min & Centro de gravedad idéntico en ambos grupos. \\
\textbf{Mediana ($\tilde{x}$)} & 10.0 min & 10.0 min & El valor central de la serie ordenada es 10. \\
\textbf{Moda ($\hat{x}$)} & 10.0 min & 10.0 min & La frecuencia máxima corresponde a 10 min. \\
\bottomrule
\end{tabularx}
\vspace{0.3cm}
\begin{itemize}\setlength{\itemsep}{0.15cm}
	\item \textbf{Ilusión de Homogeneidad:} Quedarse solo en la tendencia central enmascara el comportamiento real de los procesos.
	\item \textbf{Necesidad Epistémica:} Requerimos métricas que cuantifiquen la \textbf{separación o dispersión} respecto al valor central.
\end{itemize}
\end{frame}

% SLIDE 04: MEDIDAS DE DISPERSION Y DESVIACION MEDIA
\begin{frame}{Fases 2 y 3 STEAM · Medidas de Dispersión y Desviación Media}
\small
\begin{columns}[T]
	\begin{column}{0.46\textwidth}
		\textbf{Definición Formal de Desviación Media ($DM$):}
		\begin{equation*}
			DM = \frac{\sum_{i=1}^{n} |x_i - \bar{x}|}{n}
		\end{equation*}
		\vspace{-0.2cm}
		\begin{itemize}\setlength{\itemsep}{0.1cm}
			\item Distancia promedio de los datos a la media.
			\item Emplea valor absoluto para evitar anulación.
			\item No confundir con varianza ni desviación estándar.
		\end{itemize}
	\end{column}
	\begin{column}{0.52\textwidth}
		\renewcommand{\arraystretch}{1.15}
		\begin{tabularx}{\linewidth}{l|c|c}
			\toprule
			\rowcolor{iiiepePale} \textbf{Parámetro} & \textbf{Proceso A} & \textbf{Proceso B} \\
			\midrule
			Rango ($x_{\max}-x_{\min}$) & $12 - 8 = \mathbf{4}$ min & $15 - 5 = \mathbf{10}$ min \\
			Distancias $|x_i - 10|$ & $2, 1, 0, 0, 1, 2$ & $5, 3, 0, 0, 3, 5$ \\
			Suma de distancias & $\sum = 6$ & $\sum = 16$ \\
			\textbf{Desviación Media ($DM$)} & $\mathbf{1.0}$ \textbf{min} & $\frac{16}{6} = \mathbf{2.67}$ \textbf{min} \\
			\bottomrule
		\end{tabularx}
	\end{column}
\end{columns}
\vspace{0.3cm}
\centering
\colorbox{iiiepePale}{\textbf{Hallazgo Matemático:} La variabilidad del Proceso B es casi el \textbf{triple} que la del Proceso A.}
\end{frame}

% SLIDE 09: RETO DE TRANSFERENCIA Y SENSIBILIDAD
\begin{frame}{Reto de Transferencia y Prueba de Sensibilidad ante Outliers}
\small
\textbf{Situación Imprevista:} En el Proceso A, una falla mecánica inesperada eleva el sexto ajuste de 12 a \textbf{30 minutos}. Datos nuevos: $8, 9, 10, 10, 11, \mathbf{30}$.
\vspace{0.2cm}
\begin{columns}[T]
	\begin{column}{0.48\textwidth}
		\renewcommand{\arraystretch}{1.15}
		\begin{tabularx}{\linewidth}{l|c|c}
			\toprule
			\rowcolor{iiiepePale} \textbf{Medida} & \textbf{Antes ($x_6=12$)} & \textbf{Después ($x_6=30$)} \\
			\midrule
			Media ($\bar{x}$) & 10.0 min & $\mathbf{13.0}$ \textbf{min} ($+30\%$) \\
			Mediana ($\tilde{x}$) & 10.0 min & $\mathbf{10.0}$ \textbf{min} (\textbf{Robusta}) \\
			Rango ($R$) & 4.0 min & $\mathbf{22.0}$ \textbf{min} ($+450\%$) \\
			Desv. Media ($DM$) & 1.0 min & $\mathbf{5.67}$ \textbf{min} ($\times 5.6$) \\
			\bottomrule
		\end{tabularx}
	\end{column}
	\begin{column}{0.48\textwidth}
		\begin{alertblock}{Conclusión Matemática Crucial}
			\begin{itemize}\setlength{\itemsep}{0.1cm}
				\item La \textbf{mediana es robusta} ante valores atípicos; preserva la centralidad del conjunto.
				\item La \textbf{media y la desviación media} son altamente sensibles a valores extremos (outliers).
				\item Enseña al alumno a detectar eventos críticos en sistemas de mantenimiento.
			\end{itemize}
		\end{alertblock}
	\end{column}
\end{columns}
\end{frame}

% SLIDE 10: EVALUACION FORMATIVA Y FICHA DIDACTICA
\begin{frame}{Estrategia de Evaluación Formativa y Ficha de Trabajo}
\small
\begin{columns}[T]
	\begin{column}{0.50\textwidth}
		\textbf{Rúbrica de Desempeño (4 Niveles):}
		\begin{enumerate}\setlength{\itemsep}{0.1cm}
			\item \textbf{Destacado:} Calcula medidas con precisión, vincula gráfica y tabla, justifica con criterios técnicos y evalúa outliers.
			\item \textbf{Autónomo:} Obtiene medidas de centro y dispersión y argumenta su recomendación.
			\item \textbf{Básico:} Calcula la media pero confunde dispersión con tendencia central.
			\item \textbf{Inicial:} Requiere andamiaje en la sumatoria modular y el cálculo de la media.
		\end{enumerate}
	\end{column}
	\begin{column}{0.46\textwidth}
		\begin{block}{Ficha del Estudiante (3 Retos)}
			\begin{itemize}\setlength{\itemsep}{0.1cm}

				\item \textbf{Reto 2:} Elaboración gráfica e informe de recomendación técnica.
				\item \textbf{Reto 3:} Prueba de sensibilidad con outlier de 30 min y justificación.
