\begin{frame}{Proyectos comunitarios}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Investigar una situación y construir una respuesta con destinatarios.
\item Docente: acompaña plan y criterios; estudiante: decide, produce y revisa.
\item Palabras clave: comunidad, contexto, producción y comunicación.
\end{itemize}
\end{frame}
\begin{frame}{Indagación con STEAM}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Preguntar, diseñar indagación, organizar, comunicar y reflexionar.
\item Integrar disciplinas por su función en el problema.
\item Palabras clave: hipótesis, evidencia, modelación y diseño.
\end{itemize}
\end{frame}
\begin{frame}{Aprendizaje basado en problemas}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Recuperar saberes, definir el problema, investigar y contrastar soluciones.
\item Docente: orienta y pide razones; estudiante: argumenta y revisa.
\item Distinguir problemas de proyectos al utilizar la sigla ABP.
\end{itemize}
\end{frame}
\begin{frame}{Aprendizaje servicio}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Vincular un aprendizaje con una necesidad comunitaria acordada.
\item Escuchar a destinatarios, actuar y evaluar reciprocidad.
\item No reducir el servicio a una actividad asistencial aislada.
\end{itemize}
\end{frame}
\begin{frame}{Propuesta para mi contexto}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Indagación sobre un sistema de medición de bajo costo.
\item En preparatoria: proporcionalidad; en ingeniería: calibración y error.
\item Evaluar variables, datos, interpretación y uso de la devolución.
\end{itemize}
\end{frame}
