\begin{frame}{Cuatro pilares complementarios}
\small\renewcommand{\arraystretch}{1.45}\begin{tabularx}{\linewidth}{X|X|X}\toprule
\textbf{Pilar} & \textbf{Acción matemática} & \textbf{Evidencia}\\\midrule
Conocer & Interpretar pendiente y cuota fija. & Explicación del modelo.\\\addlinespace
Hacer & Resolver y comprobar la igualdad. & Procedimiento trazable.\\\addlinespace
Ser & Decidir y declarar supuestos. & Recomendación responsable.\\\addlinespace
Vivir juntos & Contrastar y revisar estrategias. & Corrección argumentada.\\\addlinespace
\bottomrule\end{tabularx}\par\vspace{.25cm}\small Fuente: Scott (2015b); aplicación didáctica propia.
\end{frame}
\begin{frame}{Aprender a conocer}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Comprender relaciones, formular preguntas y examinar información.
\item En calibración: identificar variables, unidades y supuestos.
\item Evidencia: justificar por qué un modelo representa el problema.
\end{itemize}
\end{frame}
\begin{frame}{Aprender a hacer}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Utilizar y revisar conocimientos en una situación.
\item Diseñar un procedimiento y calcular e interpretar error.
\item Evidencia: registro trazable y una decisión justificada.
\end{itemize}
\end{frame}
\begin{frame}{Aprender a ser}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Autonomía, responsabilidad y juicio.
\item Reconocer límites, pedir apoyo y decidir con criterios.
\item Autonomía no equivale a trabajar sin acompañamiento.
\end{itemize}
\end{frame}
\begin{frame}{Aprender a vivir juntos}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Escuchar, contrastar y construir acuerdos.
\item Rotar tareas técnicas y argumentativas.
\item Evidencia: decisiones compartidas y revisión respetuosa.
\end{itemize}
\end{frame}
\begin{frame}{Aplicación y conclusión}
\fontsize{14}{18}\selectfont\begin{itemize}\setlength{\itemsep}{.45cm}
\item Comparar dos procesos: interpretar datos, calcular y recomendar.
\item Ofrecer apoyos diferenciados con un criterio común.
\item La precisión técnica se integra con formación personal y convivencia.
\end{itemize}
\end{frame}
